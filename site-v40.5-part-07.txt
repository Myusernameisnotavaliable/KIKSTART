  function V22_routeGameKey(e,isDown){
    if(V22_keyCapture&&isDown){
      e.preventDefault();e.stopImmediatePropagation();
      if(e.code==="Escape"){V22_keyCapture=null;V22_renderHub();return;}
      const skillKeys=(window.KIKSTART_V32_SETTINGS&&window.KIKSTART_V32_SETTINGS.skillKeys)||{};
      if(Object.values(skillKeys).includes(e.code)){V22_keyCapture=null;V22_renderHub();V22_hubToast("⌨️","KEY USED BY A SKILL","Choose a key that is not assigned to a skill move.");return;}
      const taken=Object.keys(V22_bindings).find(k=>k!==V22_keyCapture&&V22_bindings[k]===e.code);
      if(taken){V22_keyCapture=null;V22_renderHub();V22_hubToast("⌨️","KEY ALREADY IN USE","Choose a different key for this control.");return;}
      V22_bindings[V22_keyCapture]=e.code;V22_keyCapture=null;V22_storeSettings();V22_renderHub();return;
    }
    const mapped=Object.keys(V22_bindings).find(k=>V22_bindings[k]===e.code);
    const oldDefault=Object.keys(V22_defaultBindings).find(k=>V22_defaultBindings[k]===e.code)||Object.keys(V22_defaultAliases).find(k=>(V22_defaultAliases[k]||[]).includes(e.code));
    if(mapped&&V22_bindings[mapped]!==V22_defaultBindings[mapped]){
      e.preventDefault();e.stopImmediatePropagation();K[V22_remapCodes[mapped]]=isDown?1:0;return;
    }
    if(oldDefault&&V22_bindings[oldDefault]!==V22_defaultBindings[oldDefault]){e.preventDefault();e.stopImmediatePropagation();return;}
  }

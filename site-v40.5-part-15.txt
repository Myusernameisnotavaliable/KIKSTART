    const result=previousKickoff.apply(this,arguments);

      const directions={};
      [0,1].forEach(t=>{
        const keeper=m.pl.find(p=>p && p.t===t && p.gk);
        if(keeper && Number.isFinite(keeper.x) && Math.abs(keeper.x)>1){
          directions[t]=-Math.sign(keeper.x);

  const previousShoot=shoot;
  shoot=function(player,power,aimZ,shotHeight){
    const m=M;
    if(!m || !player || !m.F || !Array.isArray(m.pl)){
      return previousShoot.apply(this,arguments);
    }
    const keeper=m.pl.find(p=>p && p.t===player.t && p.gk);
    let attackDir=m.ksAttackDirectionByTeam && m.ksAttackDirectionByTeam[player.t];
    if(attackDir!==1 && attackDir!==-1){
      attackDir=keeper && Number.isFinite(keeper.x) && Math.abs(keeper.x)>1
        ? -Math.sign(keeper.x)
        : (player.t?-1:1)*(m.half?-1:1);
    }
    const F=m.F;
    const targetX=attackDir*F.L/2;
    const targetZ=aimZ!==undefined?aimZ:(rnd()*2-1)*F.gw*.42;
    const angle=Math.atan2(targetZ-player.z,targetX-player.x)+
      (rnd()-.5)*(1.25-player.o/100)*.3/player.sk;
    const speed=24+power*14;
    const height=cl(Number.isFinite(shotHeight)?shotHeight:(player.t?.35:(m.aimHeight??.35)),0,1);
    const lift=.4+height*4.8+power*.35;
    kick(player,Math.cos(angle)*speed,Math.sin(angle)*speed,lift);
  };

  function NET_pickRemoteCtrl(){
    const m=M;if(!m||!m.ball)return null;
    if(m.owner&&m.owner.t===1&&!m.owner.gk){m.remoteCtrl=m.owner;return m.remoteCtrl;}

    m.remoteCtrl=best;return best;

  function NET_remotePass(p,ix,iz,len){
    let best=null,bs=-1e9;
    M.pl.forEach(q=>{
      if(q.t!==1||q===p||q.gk)return;
      const dx=q.x-p.x,dz=q.z-p.z,d=hyp(dx,dz);if(d<3*M.F.sc)return;
      const attackDir=(p.t?-1:1)*(M.half?-1:1);let score=-d*.08+(len?(dx*ix+dz*iz)/d*10:dx*attackDir/d*6);
      M.pl.forEach(o=>{if(o.t===0&&hyp(o.x-q.x,o.z-q.z)<4*M.F.sc)score-=4;});
      if(score>bs){bs=score;best=q;}
    });
    if(best)passTo(p,best);
  }

  function NET_applySnapshot(snapshot,dt){
    const m=M;if(!m||m.onlineRole!=="guest"||!snapshot||Number(snapshot.seq)<NET.lastAppliedSeq)return;
    if(!Array.isArray(snapshot.players)||snapshot.players.length!==m.pl.length||!Array.isArray(snapshot.ball))return;
    // Re-apply the latest target every render frame so the guest smoothly converges
    // between network packets instead of stepping once per packet at 30 Hz.
    NET.lastAppliedSeq=Math.max(NET.lastAppliedSeq,Number(snapshot.seq)||0);
    const k=1-Math.exp(-Math.max(.001,dt||.016)*28),n=m.F.n;
    snapshot.players.forEach((state,hostIndex)=>{
      const localIndex=NET_mapHostIndex(hostIndex,n),p=m.pl[localIndex];if(!p||!Array.isArray(state))return;
      const tx=-Number(state[0]||0),tz=-Number(state[1]||0);

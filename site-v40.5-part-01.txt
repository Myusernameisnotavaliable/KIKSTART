  const m=M,F=m.F,b=m.ball,o=m.owner,a=(p.t?-1:1)*(m.half?-1:1),h=home(p);let tx=h[0],tz=h[1],spd=p.spd*(p.t?p.sk:1);
  if(p.gk){
    const gx=a*(-F.L/2+1.8*F.sc);tx=gx;tz=cl(b.z*.55,-F.gw*.5,F.gw*.5);const bd=hyp(b.x-gx,b.z);
    if((!o||o.t!==p.t)&&bd<9*F.sc){tx=gx+a*Math.min(bd,6*F.sc)*.6;tz=cl(b.z,-F.gw*.6,F.gw*.6)}spd*=1.25;
  }else if(o===p){
    const gxp=a*F.L/2;tx=gxp;tz=p.z*.6;p.dec-=dt;
    if(p.dec<=0){p.dec=.4/p.sk;const gd=hyp(gxp-p.x,p.z),pr=nearOpp(p);
      if(gd<F.L*.28&&rnd()<.55*p.sk)shoot(p,rnd());else if(pr<3.2*F.sc&&rnd()<.65)aiPass(p);else if(rnd()<.1)aiPass(p)}
  }else if(o&&o.t===p.t){tx=h[0]+(b.x-h[0])*.3+a*F.L*.08;tz=h[1]+(b.z-h[1])*.25}
  else if((m["c"+p.t]||[]).includes(p)){tx=b.x+b.vx*.25;tz=b.z+b.vz*.25;if(o){tx=o.x;tz=o.z}}
  else{tx=h[0]+(b.x-h[0])*.35;tz=h[1]+(b.z-h[1])*.35}
  const dx=tx-p.x,dz=tz-p.z,d=hyp(dx,dz);let sx=0,sz=0;if(d>.6){sx=dx/d;sz=dz/d}
  move(p,sx,sz,spd*Math.min(1,d/2.5)*(p.lunge>0?1.4:1),dt);if(d>.6){p.dx=sx;p.dz=sz}
}
function ballStep(dt){
  const m=M,b=m.ball,F=m.F,o=m.owner;b.lock-=dt;
  if(o){
    b.x=o.x+o.dx*1.1*F.sc;b.z=o.z+o.dz*1.1*F.sc;b.y=.5*F.ps;b.vy=0;b.vx=o.vx;b.vz=o.vz;
    m.pl.forEach(q=>{
      if(q.t===o.t||q.cool>0)return;const d=hyp(q.x-o.x,q.z-o.z);
      if(d<(q.lunge>0?2.4:1.7)*F.sc){
        q.cool=q.lunge>0?.35:.55;const ch=.32*(q.o+25)/(o.o+25)*q.sk*(q.lunge>0?1.7:1)*(o===m.ctrl?.85:1);
        if(rnd()<ch){m.owner=q;b.lock=.2;m.lastT=q.t;o.cool=.7;q.dec=.2;q.hold=0;sfx.pop()}
      }
    });
    if(m.owner&&m.owner.gk){m.owner.hold+=dt;if(m.owner.hold>1.2)aiPass(m.owner)}
  }else{
    b.x+=b.vx*dt;b.z+=b.vz*dt;
    b.y+=b.vy*dt;b.vy-=6.8*dt;
    const floor=.5*F.ps;
    if(b.y<floor){b.y=floor;if(Math.abs(b.vy)>1.15)b.vy=-b.vy*.22;else b.vy=0;}
    const f=Math.pow(.5,dt);b.vx*=f;b.vz*=f;
    let best=null,bd=1e9;const sp=hyp(b.vx,b.vz);
    m.pl.forEach(p=>{const reach=(p.gk?2.4:1.5)*F.sc,d=hyp(p.x-b.x,p.z-b.z),maxHeight=(p.gk?2.7:1.65)*F.ps;if(b.y<=maxHeight&&d<reach&&d<bd&&(b.lock<=0||p.t!==m.lastT)&&(p.gk||sp<22)){bd=d;best=p}});
    if(best){m.owner=best;best.hold=0;m.lastT=best.t;b.vx=b.vy=b.vz=0;b.y=floor}
    else if(Math.abs(b.x)>F.L/2){if(b.y<2.65*F.ps&&Math.abs(b.z)<F.gw/2)goal(b.x>0?0:1);else out()}
    else if(Math.abs(b.z)>F.W/2)out();
  }
}
function goal(t){
  const m=M;m.score[t]++;m.pause=5.2;m.scorer=m.lastP;m.celeb=1;m.ball.vx=m.ball.vy=m.ball.vz=0;m.ball.y=.5*m.F.ps;m.owner=null;banner(t===0?"GOAL! ⚽":"They scored…");
  sfx.whistle();sfx.horn();cheer(t===0?5+S.st.a*.5:2,t===0?.5+.06*S.st.a:.2);boom(.5);if(t===0)burst(innerWidth/2,innerHeight/3,COL.gold.concat(COL.diamond),90,9);
  m.after=()=>kickoff(1-t);
}
function out(){
  const m=M,b=m.ball,F=m.F,t=1-m.lastT,rx=cl(b.x,-F.L/2+2,F.L/2-2),rz=cl(b.z,-F.W/2+1,F.W/2-1);
  b.x=rx;b.z=rz;b.vx=b.vy=b.vz=0;b.y=.5*F.ps;m.pause=.7;m.celeb=0;sfx.whistle();
  m.after=()=>{let best=null,bd=1e9;m.pl.forEach(p=>{if(p.t===t&&!p.gk){const d=hyp(p.x-rx,p.z-rz);if(d<bd){bd=d;best=p}}});if(best){best.x=rx-(best.t?-1:1)*1.2;best.z=rz;m.owner=best;best.hold=0;m.lastT=t}};
}
function stepMatch(dt){
  const m=M,b=m.ball;
  if(m.wait){}
  else if(m.intro>0){
    m.intro-=dt;const elapsed=5.8-m.intro;
    m.pl.forEach(p=>{const h=home(p),u=cl((elapsed-(p.walkDelay||0))/4.2,0,1),e=u*u*(3-2*u);p.x=p.sx+(h[0]-p.sx)*e;p.z=p.sz+(h[1]-p.sz)*e;p.dx=0;p.dz=1;p.vx=0;p.vz=u<1?4.4:0});
    b.x=b.z=0;
    if(m.intro<=0){kickoff(0);m.pause=1.2;banner("KICK OFF!");sfx.whistle()}
  }
  else if(m.t>=m.total){finishMatch(false);return}
  else if(m.pause>0){m.pause-=dt;if(m.pause<=0&&m.after){const f=m.after;m.after=null;f()}}
  else{
    m.t+=dt;pickCtrl();
    [0,1].forEach(t=>{const l=m.pl.filter(p=>p.t===t&&!p.gk).sort((a,c)=>hyp(a.x-b.x,a.z-b.z)-hyp(c.x-b.x,c.z-b.z));m["c"+t]=[l[0],m.F.n>6?l[1]:null]});
    m.pl.forEach(p=>{p.cool-=dt;p.lunge-=dt;if(p===m.ctrl)userCtrl(p,dt);else ai(p,dt)});
    ballStep(dt);
    m.amb-=dt;if(m.amb<=0){m.amb=3.5;cheer(4,(Math.abs(b.x)>m.F.L*.3?.07:.035)*(1+S.st.a*.25))}  // crowd noise rises near the goals and with a bigger stadium
  }
  V5_animate(m,dt);
  b.mesh.position.set(b.x,Math.max(.5*m.F.ps,b.y||.5*m.F.ps),b.z);b.mesh.rotation.x+=b.vz*dt*.6;b.mesh.rotation.z-=b.vx*dt*.6;b.sh.position.set(b.x,.04,b.z);b.sh.material.opacity=cl(.35/(1+Math.max(0,(b.y||.5)-.5)*.35),.08,.35);b.sh.scale.setScalar(cl(1+Math.max(0,(b.y||.5)-.5)*.18,1,1.8));

  const min=Math.min(90,Math.floor(m.t/m.total*90)),html=m.cr0+' <b>'+esc(CL[S.club].n)+'</b> <b style="font-size:24px">'+m.score[0]+' - '+m.score[1]+'</b> <b>'+esc(m.team.n)+'</b> '+m.cr1+' ⏱ '+min+"'";
  if(html!==m.hh){$("hud").innerHTML=html;m.hh=html}
}

}

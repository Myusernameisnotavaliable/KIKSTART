/* KIKSTART v26 hardening: keep code-only cards out of market restocks and improve camera targeting. */
(function KIKSTART_V26_HARDENING(){
  if (window.__kikstartV26Hardening) return;
  window.__kikstartV26Hardening = true;

  // Hidden/exclusive players have no ordinary-market rarity. Never let a secret card
  // crash a normal market restock when its rarity is not present in ST.
  const previousRestock = restock;
  restock = function safeRestock(){
    if (!S) return;
    const stock = { fb: {}, fs: {} };
    ["fb", "fs"].forEach(mode => {
      (POOL[mode] || []).forEach(player => {
        if (!player || player.secret || player.academy || player.youthGenerated || player.removed || player.messiExclusive || player.khizarExclusive || player.daniyalExclusive) {
          if (player) stock[mode][player.id] = 0;
          return;
        }
        const odds = ST && ST[player.r];
        if (!odds || !Number.isFinite(odds[0]) || !Number.isFinite(odds[1])) {
          stock[mode][player.id] = 0;
          return;
        }
        if (rnd() < odds[0]) stock[mode][player.id] = 1 + Math.floor(rnd() * odds[1]);
      });
    });
    S.stock = stock;
    S.rest = Date.now() + 120000;
    save();
  };

  // Explicitly initialise stock safely for an already-loaded save too.
  if (S) {
    try { restock(); }
    catch (error) { console.warn("KIKSTART safe market restock:", error); }
  }
})();

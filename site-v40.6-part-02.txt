// Player pack odds now select players individually; star forwards are deliberately very rare.
function V27_weightedPlayer(pool){
  // Ordinary cards are the baseline. Icons and Heroes are substantially rarer;
  // the headline legends are exceptionally rare even after a high-tier roll.
  const headline=new Set(["Lionel Messi","Cristiano Ronaldo","Neymar","Diego Maradona","Pelé","Zinedine Zidane","Ronaldinho","Paolo Maldini"]);
  const eligible=pool.filter(p=>p&&!p.secret&&!p.messiExclusive&&!p.daniyal&&!p.khizargive&&!p.shrekCard&&!p.allTimeXI&&!p.codeExclusive&&!p.launchEventKey&&!p.academy&&!p.removed&&!p.v32Ayden&&p.kind!=="secret");
  if(!eligible.length)throw new Error("No eligible player cards for this pack tier.");
  const weights=eligible.map(p=>({p,w:headline.has(p.n)?.006:(p.kind==="icon"?.045:(p.kind==="hero"?.12:1))}));
  let total=weights.reduce((a,x)=>a+x.w,0),r=rnd()*total;
  for(const x of weights){r-=x.w;if(r<=0)return x.p;}
  return weights[weights.length-1].p;
}

function V27_roofStrip(axis,sign,inner,outer,length,y,material){
  const along=24,across=8,pos=[],idx=[];
  for(let i=0;i<=along;i++){
    const u=-length/2+(length*i/along);
    for(let j=0;j<=across;j++){
      const v=j/across,span=inner+(outer-inner)*v;
      const arch=Math.sin(v*Math.PI)*2.25;
      const yy=y+arch-v*.45;
      if(axis==="x")pos.push(u,yy,sign*span);else pos.push(sign*span,yy,u);
    }
  }
  for(let i=0;i<along;i++)for(let j=0;j<across;j++){
    const a=i*(across+1)+j,b=a+across+1;idx.push(a,b,a+1,b,b+1,a+1);
  }
  const g=new THREE.BufferGeometry();g.setAttribute("position",new THREE.Float32BufferAttribute(pos,3));g.setIndex(idx);g.computeVertexNormals();
  const mesh=new THREE.Mesh(g,material);mesh.castShadow=true;mesh.receiveShadow=true;sw.add(mesh);

  const targetZ=aimz!==undefined?aimz:(rnd()*2-1)*F.gw*.42;
  const ang=Math.atan2(targetZ-p.z,attack*F.L/2-p.x)+(rnd()-.5)*(1.25-p.o/100)*.3/(p.sk||1);

  // Smooth curve makes the lower half useful for driven shots and the upper half for chips.
  const lift=.85+Math.pow(height,1.08)*4.9+power*.28;

};

  function K16_initStadiumWhenReady() {
    const start = () => {
      if (typeof THREE === "undefined") throw new Error("Three.js is not ready");
      // The v14 stadium installer is asynchronous when Three.js comes from a CDN.
      // Re-apply the v16 builder after it has completed so its generic builder cannot overwrite ours.
      K16_installStadiumOverrides();
      const element = $("stad3d");
      if (!element) return;
      try {
        initStad();
        if (sr && sr.domElement && sr.domElement.parentElement !== element) element.appendChild(sr.domElement);
        if (sr && sc) {
          const width = Math.max(1, element.clientWidth);
          const height = Math.max(1, element.clientHeight);
          sr.setSize(width, height);
          sc.aspect = width / height;
          sc.updateProjectionMatrix();
        }
        buildStad();
      } catch (error) {
        console.warn("KIKSTART stadium initialisation:", error);
        const note = $("k16StadiumStatus");
        if (note) note.textContent = "The 3D stadium could not initialise. Refresh the page and try again.";
      }
    };
    if (typeof THREE !== "undefined") { start(); return; }
    const promise = window.__kikThreePromise;
    if (promise && typeof promise.then === "function") {
      promise.then(start).catch(error => {
        console.warn("KIKSTART 3D library unavailable:", error);
        const note = $("k16StadiumStatus");
        if (note) note.textContent = "The 3D library is blocked or unavailable. Check your connection and refresh.";
      });
    } else {
      const note = $("k16StadiumStatus");
      if (note) note.textContent = "The 3D library is still loading. Try opening Stadium again in a moment.";
    }
  }

  function K16_dispose(group) {
    if (!group) return;
    group.traverse(object => {
      if (object.geometry && object.geometry.dispose) object.geometry.dispose();
      if (object.material) (Array.isArray(object.material) ? object.material : [object.material]).forEach(material => {
        if (material && material.map && material.map.dispose) material.map.dispose();
        if (material && material.dispose) material.dispose();
      });
    });
